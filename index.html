function dayWithExercises(db, dayId) {
 const day = db.workout_days.find(d => d.id === dayId);
 if (!day) return null;
 const des = db.day_exercises
 .filter(de => de.day_id === dayId)
 .slice().sort((a, b) => a.position - b.position)
 .map(de => ({ ...de, exercise: db.exercises.find(e => e.id === de.exercise_id) }))
 .filter(de => de.exercise);
 const ss = db.supersets.filter(s => s.day_id === dayId).slice().sort((a, b) => a.position - b.position);
 return { ...day, exercises: des, supersets: ss };
 }

function getNextPos(rows) {
 if (!rows.length) return 1;
 return Math.max(...rows.map(r => r.position || 0)) + 1;
 }

function getExerciseReliableGroupName(db, exerciseId) {
 const relatedDayExercises = (db.day_exercises || []).filter(de => de.exercise_id === exerciseId);
 if (!relatedDayExercises.length) return '';

const seenGroupIds = new Set();
 const seenGroupNames = new Set();

for (const de of relatedDayExercises) {
 const day = (db.workout_days || []).find(d => d.id === de.day_id);
 if (!day || !day.group_id) return '';
 const group = (db.groups || []).find(g => g.id === day.group_id);
 if (!group || !group.name) return '';
 seenGroupIds.add(group.id);
 seenGroupNames.add(group.name.trim());
 if (seenGroupIds.size > 1 || seenGroupNames.size > 1) return '';
 }

return seenGroupNames.size === 1 ? Array.from(seenGroupNames) : '';
 }

function parseBulkImport(text) {
 const lines = text.split(/\r?\n/);
 const rows = [];
 let currentDay = '';
 let currentGroup = '';

for (const raw of lines) {
 const line = raw.trim();
 if (!line) continue;

if (/^DAY:/i.test(line)) {
 currentDay = line.replace(/^DAY:/i, '').trim();
 continue;
 }
 if (/^GROUP:/i.test(line)) {
 currentGroup = line.replace(/^GROUP:/i, '').trim();
 continue;
 }
 if (/^EXERCISE:/i.test(line)) {
 const body = line.replace(/^EXERCISE:/i, '');
 const parts = body.split('|').map(s => s.trim());
 const exName = parts.trim();
 if (!exName) continue;

const kv = {};
 for (let i = 1; i < parts.length; i++) {
 const eq = parts[i].indexOf('=');
 if (eq === -1) continue;
 const k = parts[i].slice(0, eq).trim().toLowerCase();
 const v = parts[i].slice(eq + 1).trim();
 kv[k] = v;
 }

const isBand = 'band' in kv && kv['band'] !== '';
 const hasDuration = 'duration' in kv && kv['duration'] !== '';
 const kind = hasDuration ? 'time' : 'reps';
 const isMain = (kv['main'] || '').toLowerCase() === 'true';
 const isWarmup = (kv['warmup'] || '').toLowerCase() === 'true';

if (!currentDay) continue;

rows.push({
 dayName: currentDay,
 groupName: currentGroup,
 exName,
 sets: kv['sets'] || null,
 reps: kv['reps'] || null,
 weight: isBand ? null : (kv['weight'] || null),
 bandLabel: isBand ? kv['band'] : null,
 duration: hasDuration ? (parseInt(kv['duration']) || 30) : null,
 rest: kv['rest'] || null,
 notes: kv['notes'] || null,
 isMain,
 isWarmup,
 isBand,
 kind,
 });
 continue;
 }
 }

if (rows.length === 0) {
 return { error: 'No valid EXERCISE lines found. Make sure each exercise starts with "EXERCISE:" and is inside a "DAY:" block.' };
 }

const dayNames = [...new Set(rows.map(r => r.dayName))];
 const exNames = [...new Set(rows.map(r => r.exName))];
 return { rows, dayNames, exNames, error: null };
 }

function applyBulkImport(db, rows) {
 const d = db;

const groupsByName = new Map();
 d.groups.forEach(g => groupsByName.set(g.name.toLowerCase(), g));
 const getOrCreateGroup = (name) => {
 if (!name) return null;
 const key = name.toLowerCase();
 if (groupsByName.has(key)) return groupsByName.get(key).id;
 const id = uuid();
 const g = { id, name, sort_order: getNextPos(d.groups), created_at: nowSec() };
 d.groups.push(g);
 groupsByName.set(key, g);
 return id;
 };

const exByName = new Map();
 d.exercises.forEach(e => exByName.set(e.name.toLowerCase(), e));
 const getOrCreateExercise = (row) => {
 const key = row.exName.toLowerCase();
 if (exByName.has(key)) return exByName.get(key).id;
 const id = uuid();
 const ex = {
 id,
 name: row.exName,
 kind: row.kind,
 use_band: row.isBand ? 1 : 0,
 bilateral: 0,
 description: '',
 persistent_notes: '',
 youtube_url: '',
 created_at: nowSec(),
 archived: 0,
 };
 d.exercises.push(ex);
 exByName.set(key, ex);
 return id;
 };

const daysByName = new Map();
 d.workout_days.forEach(day => daysByName.set(day.name.toLowerCase(), day));
 const getOrCreateDay = (name, groupId) => {
 const key = name.toLowerCase();
 if (daysByName.has(key)) return daysByName.get(key).id;
 const id = uuid();
 const day = { id, name, sort_order: getNextPos(d.workout_days), created_at: nowSec(), group_id: groupId };
 d.workout_days.push(day);
 daysByName.set(key, day);
 return id;
 };

for (const row of rows) {
 const groupId = getOrCreateGroup(row.groupName);
 const dayId = getOrCreateDay(row.dayName, groupId);
 const exId = getOrCreateExercise(row);
 const existingInDay = d.day_exercises.filter(de => de.day_id === dayId);
 d.day_exercises.push({
 id: uuid(),
 day_id: dayId,
 exercise_id: exId,
 position: getNextPos(existingInDay),
 planned_sets: row.sets ? (parseInt(row.sets) || 1) : 3,
 planned_reps: row.kind === 'reps' ? (row.reps || null) : null,
 planned_weight: row.isBand ? (row.bandLabel || null) : (row.weight || null),
 planned_duration_sec: row.kind === 'time' ? (row.duration || 30) : null,
 notes: row.notes || null,
 rest_sec: row.rest || null,
 is_warmup: row.isWarmup ? 1 : 0,
 is_main: row.isMain ? 1 : 0,
 superset_id: null,
 created_at: nowSec(),
 });
 }

return d;
 }

function getExerciseDefaultsMap(db) {
 const map = new Map();
 const finished = (db.sessions || [])
 .filter(s => s.is_finished)
 .slice().sort((a, b) => (b.finished_at || 0) - (a.finished_at || 0));
 for (const s of finished) {
 const ses = (db.session_exercises || []).filter(se => se.session_id === s.id);
 for (const se of ses) {
 if (map.has(se.exercise_id)) continue;
 const sets = (db.session_sets || [])
 .filter(ss => ss.session_exercise_id === se.id && ss.completed)
 .sort((a, b) => a.set_number - b.set_number);
 const lastSet = sets[sets.length - 1];
 map.set(se.exercise_id, {
 reps: (lastSet?.actual_reps ?? se.planned_reps) || null,
 weight: (lastSet?.actual_weight ?? se.planned_weight) || null,
 duration_sec: (lastSet?.actual_duration_sec ?? se.planned_duration_sec) || null,
 });
 }
 } return map;
 }

function playBeep() {
 try {
 const ctx = new (window.AudioContext || window.webkitAudioContext)();
 [[880, 0.55], [1320, 0.30], [440, 0.20]].forEach(([freq, vol]) => {
 const o = ctx.createOscillator(); const g = ctx.createGain();
 o.connect(g); g.connect(ctx.destination);
 o.type = 'sine'; o.frequency.value = freq;
 g.gain.setValueAtTime(0.001, ctx.currentTime);
 g.gain.exponentialRampToValueAtTime(vol, ctx.currentTime + 0.02);
 g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.55);
 o.start(); o.stop(ctx.currentTime + 0.6);
 });
 setTimeout(() => ctx.close(), 900);
 } catch (e) {}
 }

 const existing = d.session_sets.filter(s => s.session_exercise_id === se.id).sort((a, b) => a.set_number - b.set_number);
 const current = existing.length;
 const next = Math.max(1, current + delta);
 if (next === current) return d;
 if (next > current) {
 for (let i = current + 1; i <= next; i++) {
 d.session_sets.push({ id: uuid(), session_exercise_id: se.id, set_number: i, actual_reps: null, actual_weight: null, completed: 0, actual_duration_sec: null });
 }
 } else {
 const removable = [...existing].reverse().filter(s => !s.completed);
 const toRemove = removable.slice(0, current - next).map(s => s.id);
 if (toRemove.length < current - next) {
 const completed = [...existing].reverse().filter(s => s.completed);
 const extra = completed.slice(0, (current - next) - toRemove.length).map(s => s.id);
 toRemove.push(...extra);
 }
 const removeSet = new Set(toRemove);
 d.session_sets = d.session_sets.filter(s => !removeSet.has(s.id));
 d.session_sets.filter(s => s.session_exercise_id === se.id).sort((a, b) => a.set_number - b.set_number).forEach((s, i) => { s.set_number = i + 1; });
 }
 const se_row = d.session_exercises.find(x => x.id === se.id);
 if (se_row) se_row.planned_sets = next;
 return d;
 });
 };

if (!ex) return null;
 const isTime = se.kind === 'time';
 const isBilateral = isTime && !!ex.bilateral;
 const useBand = !isTime && !!ex.use_band;
 const loadPlaceholder = useBand ? 'band' : 'weight';

return (

 <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M17 1l4 4-4 4"/><path d="M3 11V9a4 4 0 0 1 4-4h14"/><path d="M7 23l-4-4 4-4"/><path d="M21 13v2a4 4 0 0 1-4 4H3"/></svg>
 </button>
 <button onClick={toggleCompleted} className={h-9 w-9 rounded-full flex items-center justify-center ${se.completed ? 'bg-brand text-white' : 'bg-neutral-100 text-neutral-500'}} aria-label="Toggle complete">{I.check}</button>
 </div>
 </div>

{ex.persistent_notes && <div className="text-xs text-neutral-600 italic bg-amber-50 border border-amber-200 rounded-md p-2 mb-2">{ex.persistent_notes}</div>}
 {se.notes_snapshot && <div className="text-xs text-neutral-600 bg-neutral-50 rounded-md p-2 mb-2">{se.notes_snapshot}</div>}

{isTime ? (
 <div className="space-y-2">

 <Label className="mb-0 text-xs text-neutral-500">Duration (s)</Label>

 ) : (

 );
 }

function TimeSetRow({ s, plannedSec, updateSet, onComplete }) {
 const [running, setRunning] = useState(false);
 const [remaining, setRemaining] = useState(plannedSec);
 const tickRef = useRef(null);
 const plannedRef = useRef(plannedSec);

useEffect(() => { plannedRef.current = plannedSec; }, [plannedSec]);
 useEffect(() => () => { if (tickRef.current) clearInterval(tickRef.current); }, []);
 useEffect(() => { if (!running) setRemaining(plannedSec); }, [plannedSec, running]);

const start = () => {
 const startPlanned = plannedRef.current;
 setRemaining(startPlanned);
 setRunning(true);
 const startedAt = Date.now();
 tickRef.current = setInterval(() => {
 const elapsed = Math.floor((Date.now() - startedAt) / 1000);
 const left = startPlanned - elapsed;
 if (left <= 0) {
 clearInterval(tickRef.current); tickRef.current = null;
 setRemaining(0); setRunning(false);
 try { playBeep(); } catch (e) {}
 updateSet(s.id, { actual_duration_sec: startPlanned, completed: 1 });
 if (onComplete) setTimeout(onComplete, 60);
 } else {
 setRemaining(left);
 }
 }, 250);
 };

const stop = () => {
 if (tickRef.current) { clearInterval(tickRef.current); tickRef.current = null; }
 setRunning(false);
 setRemaining(plannedRef.current);
 };

const manualToggle = () => {
 if (s.completed) {
 updateSet(s.id, { completed: 0, actual_duration_sec: null });
 } else {
 updateSet(s.id, { completed: 1, actual_duration_sec: plannedRef.current });
 if (onComplete) setTimeout(onComplete, 60);
 }
 };

return (
 <div className="flex items-center gap-2">
 <div className="w-6 text-xs text-neutral-500">#{s.set_number}</div>
 <div className="flex-1 text-sm font-mono" data-testid={text-timer-${s.id}}>
 {s.completed ? ${s.actual_duration_sec || plannedSec}s ✓ : (running ? ${remaining}s : ${plannedSec}s)}
 </div>
 {!s.completed && !running && <Button size="sm" onClick={start} data-testid={button-timer-start-${s.id}}>{I.play} Start</Button>}
 {running && <Button size="sm" variant="outline" onClick={stop} data-testid={button-timer-stop-${s.id}}>Stop</Button>}
 <button onClick={manualToggle} className={h-9 w-9 rounded-md flex items-center justify-center ${s.completed ? 'bg-brand text-white' : 'bg-neutral-100 text-neutral-500'}} data-testid={button-set-complete-${s.id}}>{I.check}</button>
 </div>
 );
 }

function BilateralTimeSetRow({ s, plannedSec, updateSet, onComplete }) {
 const [lRun, setLRun] = useState(false);
 const [rRun, setRRun] = useState(false);
 const [lRem, setLRem] = useState(plannedSec);
 const [rRem, setRRem] = useState(plannedSec);
 const [lDone, setLDone] = useState(false);
 const [rDone, setRDone] = useState(false);
 const lRef = useRef(null); const rRef = useRef(null);
 const pRef = useRef(plannedSec);
 const lDoneRef = useRef(false); const rDoneRef = useRef(false);
 useEffect(() => { pRef.current = plannedSec; }, [plannedSec]);
 useEffect(() => { lDoneRef.current = lDone; }, [lDone]);
 useEffect(() => { rDoneRef.current = rDone; }, [rDone]);
 useEffect(() => { if (!lRun && !lDone) setLRem(plannedSec); }, [plannedSec, lRun, lDone]);
 useEffect(() => { if (!rRun && !rDone) setRRem(plannedSec); }, [plannedSec, rRun, rDone]);
 useEffect(() => () => {
 if (lRef.current) clearInterval(lRef.current);
 if (rRef.current) clearInterval(rRef.current);
 }, []);

const markBothDone = () => {
 updateSet(s.id, { actual_duration_sec: pRef.current, completed: 1 });
 if (onComplete) setTimeout(onComplete, 60);
 };

const startSide = (side) => {
 const isL = side === 'L';
 const ref = isL ? lRef : rRef;
 const setRun = isL ? setLRun : setRRun;
 const setRem = isL ? setLRem : setRRem;
 const setDone = isL ? setLDone : setRDone;
 const otherDoneRef = isL ? rDoneRef : lDoneRef;
 const p = pRef.current;
 setRun(true);
 const t0 = Date.now();
 ref.current = setInterval(() => {
 const left = p - Math.floor((Date.now() - t0) / 1000);
 if (left <= 0) {
 clearInterval(ref.current); ref.current = null;
 setRem(0); setRun(false); setDone(true);
 try { playBeep(); } catch (_) {}
 if (otherDoneRef.current) markBothDone();
 } else { setRem(left); }
 }, 200);
 };

const stopSide = (side) => {
 const isL = side === 'L';
 const ref = isL ? lRef : rRef;
 const setRun = isL ? setLRun : setRRun;
 const setRem = isL ? setLRem : setRRem;
 if (ref.current) { clearInterval(ref.current); ref.current = null; }
 setRun(false); setRem(pRef.current);
 };

const manualToggle = () => {
 if (s.completed) {
 updateSet(s.id, { completed: 0, actual_duration_sec: null });
 setLDone(false); setRDone(false);
 } else {
 updateSet(s.id, { completed: 1, actual_duration_sec: pRef.current });
 setLDone(true); setRDone(true);
 if (onComplete) setTimeout(onComplete, 60);
 }
 };

const SideTimer = ({ side, running, rem, done, otherRunning }) => (
 <div className="flex items-center gap-1.5 flex-1">
 <span className={text-xs font-bold w-4 shrink-0 ${done ? 'text-brand' : 'text-neutral-400'}}>{side}</span>
 <span className="text-sm font-mono w-10 shrink-0">{done ? ${plannedSec}s✓ : (running ? ${rem}s : ${plannedSec}s)}</span>
 {!done && !running && <Button size="sm" disabled={otherRunning} onClick={() => startSide(side)}>{I.play}</Button>}
 {running && <Button size="sm" variant="outline" onClick={() => stopSide(side)}>Stop</Button>}
 </div>
 );

return (
 <div className="flex items-center gap-2">
 <div className="w-6 text-xs text-neutral-500 shrink-0">#{s.set_number}</div>
 <div className="flex-1 flex flex-col gap-1 py-0.5">
 <SideTimer side="L" running={lRun} rem={lRem} done={lDone} otherRunning={rRun} />
 <SideTimer side="R" running={rRun} rem={rRem} done={rDone} otherRunning={lRun} />
 </div>
 <button onClick={manualToggle} className={shrink-0 h-9 w-9 rounded-md flex items-center justify-center ${s.completed ? 'bg-brand text-white' : 'bg-neutral-100 text-neutral-500'}}>{I.check}</button>
 </div>
 );
 }

function CalendarPage() {

 const today = new Date();
 const [year, setYear] = useState(today.getFullYear());
 const [month, setMonth] = useState(today.getMonth());

 const m = new Map();
 db.sessions.filter(s => s.is_finished).forEach(s => {
 const dt = new Date((s.finished_at || s.started_at) * 1000);
 const key = ${dt.getFullYear()}-${dt.getMonth()}-${dt.getDate()};
 if (!m.has(key)) m.set(key, []);
 m.get(key).push(s);
 });
 return m;
 }, [db]);

const weeks = useMemo(() => {
 const first = new Date(year, month, 1);
 const last = new Date(year, month + 1, 0);
 const start = new Date(first);
 start.setDate(first.getDate() - ((first.getDay() + 6) % 7));
 const end = new Date(last);
 end.setDate(last.getDate() + (7 - ((last.getDay() + 6) % 7) - 1));
 const out = [];
 let cur = new Date(start);
 while (cur <= end) {
 const row = [];
 for (let i = 0; i < 7; i++) {
 row.push(new Date(cur));
 cur.setDate(cur.getDate() + 1);
 }
 out.push(row);
 }
 return out;
 }, [year, month]);

const ymLabel = new Date(year, month, 1).toLocaleDateString(undefined, { month: 'long', year: 'numeric' });

return (
 <PageShell title="Calendar">

 <div className="font-medium">{ymLabel}</div>
 <Button variant="outline" size="sm" onClick={() => { const d = new Date(year, month + 1, 1); setYear(d.getFullYear()); setMonth(d.getMonth()); }}>Next</Button>

 {['Mon','Tue','Wed','Thu','Fri','Sat','Sun'].map(d => <div key={d} className="p-2 text-center">{d}</div>)}
 </div>
 {weeks.map((week, idx) => (
 <div key={idx} className="grid grid-cols-7 border-b last:border-b-0 border-neutral-100">
 {week.map(d => {
 const inMonth = d.getMonth() === month;
 const key = ${d.getFullYear()}-${d.getMonth()}-${d.getDate()};
 const list = sessionsByDate.get(key) || [];
 const isToday = d.toDateString() === today.toDateString();
 return (
 <div key={key} className={min-h-[80px] p-2 border-r last:border-r-0 border-neutral-100 ${inMonth ? 'bg-white' : 'bg-neutral-50 text-neutral-400'}}>
 <div className={text-xs ${isToday ? 'font-bold text-brand' : ''}}>{d.getDate()}</div>
 <div className="mt-1 space-y-1">
 {list.slice(0, 3).map(s => {

 return (
 <button key={s.id} onClick={() => navigate(/history/${s.id})}

 {day?.name || 'Session'}

 })}

 </div>
 </div>
 );
 })}
 </div>
 ))}
 </Card>
 </div>
 </PageShell>
 );

function exportBackup(db) {
 const blob = new Blob([JSON.stringify(db, null, 2)], { type: 'application/json' });
 const a = document.createElement('a');
 a.href = URL.createObjectURL(blob);
 a.download = liftlog-backup-${new Date().toISOString().slice(0, 10)}.json;
 document.body.appendChild(a);
 a.click();
 setTimeout(() => {
 URL.revokeObjectURL(a.href);
 a.remove();
 }, 500);
 }

function SettingsPage() {
 const { db, setDb, settings, mutateSettings, showToast } = useApp();
 const fileRef = useRef(null);
 const [busy, setBusy] = useState(false);
 const [gsUrl, setGsUrl] = useState(settings.googleSheets.webAppUrl || '');
 const [sheetId, setSheetId] = useState(settings.googleSheets.spreadsheetId || '');
 const [sheetName, setSheetName] = useState(settings.googleSheets.sheetName || 'LiftLog');
 const [bulkText, setBulkText] = useState('');
 const [bulkPreview, setBulkPreview] = useState(null);
 const [bulkError, setBulkError] = useState('');
 const { confirm, node: confirmNode } = useConfirm();

useEffect(() => {
 setGsUrl(settings.googleSheets.webAppUrl || '');
 setSheetId(settings.googleSheets.spreadsheetId || '');
 setSheetName(settings.googleSheets.sheetName || 'LiftLog');
 }, [settings]);

const saveGsConfig = () => {
 mutateSettings(s => {
 s.googleSheets.webAppUrl = gsUrl.trim();
 s.googleSheets.spreadsheetId = sheetId.trim();
 s.googleSheets.sheetName = sheetName.trim() || 'LiftLog';
 return s;
 });
 showToast('Sheet settings saved');
 };

const pushToGoogleSheets = async () => {
 const webAppUrl = (gsUrl || '').trim();
 const spreadsheetId = (sheetId || '').trim();
 const targetSheetName = (sheetName || 'LiftLog').trim() || 'LiftLog';

if (!webAppUrl) {
 showToast('Add the Apps Script URL first');
 return;
 }
 if (!spreadsheetId) {
 showToast('Add the spreadsheet ID first');
 return;
 }

setBusy(true);
 try {
 const payload = {
 spreadsheetId,
 sheetName: targetSheetName,
 pushedAt: new Date().toISOString(),
 app: 'Lift Log',
 localFirst: true,
 syncMode: 'manual_push_only',
 data: db,
 };

const res = await fetch(webAppUrl, {
 method: 'POST',
 headers: { 'Content-Type': 'text/plain;charset=utf-8' },
 body: JSON.stringify(payload),
 });

const text = await res.text();
 let json = null;
 try { json = JSON.parse(text); } catch (_) {}

if (!res.ok) {
 throw new Error((json && (json.error || json.message)) || Request failed (${res.status}));
 }

if (json && json.ok === false) {
 throw new Error(json.error || json.message || 'Push failed');
 }

showToast('Local data pushed to Google Sheets');
 } catch (err) {
 console.error(err);
 showToast(err.message || 'Push failed');
 } finally {
 setBusy(false);
 }
 };

const onImportFile = async (file) => {
 if (!file) return;
 try {
 const text = await file.text();
 const parsed = JSON.parse(text);
 const base = emptyDb();
 setDb({ ...base, ...parsed, notes_library: parsed.notes_library || [] });
 showToast('Backup imported');
 } catch (e) {
 showToast('Import failed');
 }
 };

const resetAll = async () => {
 const ok = await confirm({
 title: 'Reset all local data?',
 message: 'This removes all local workout data from this browser. Export a backup first if you need one.',
 danger: true,
 confirmLabel: 'Reset',
 });
 if (!ok) return;
 const fresh = emptyDb();
 setDb(fresh);
 showToast('Local data reset');
 };

const parseNow = () => {
 const result = parseBulkImport(bulkText);
 if (result.error) {
 setBulkError(result.error);
 setBulkPreview(null);
 return;
 }
 setBulkError('');
 setBulkPreview(result);
 };

const applyNow = () => {
 if (!bulkPreview?.rows?.length) return;
 setDb(prev => applyBulkImport(structuredClone(prev), bulkPreview.rows));
 setBulkText('');
 setBulkPreview(null);
 setBulkError('');
 showToast(Imported ${bulkPreview.rows.length} entries);
 };

const configured = !!(gsUrl.trim() && sheetId.trim());

return (
 <PageShell title="Settings">
 <div className="p-4 space-y-4 pb-8">
 <Card className="p-4 space-y-3">
 <div>
 <h2 className="text-base font-semibold">Google Sheets sync</h2>
 <p className="text-sm text-neutral-600 mt-1">Local save comes first. Spreadsheet sync is optional and manual push only.</p>
 </div>

<div className="rounded-xl border border-green-200 bg-green-50 p-3 text-sm text-green-800">
 <div className="font-medium">This phone stays the source of truth.</div>
 <div className="mt-1 text-green-700">No pull sync. No auto-sync. Nothing is pushed unless you tap the button.</div>
 </div>

<div className="space-y-3">
 <div>
 <Label>Apps Script web app URL</Label>
 <Input value={gsUrl} onChange={e => setGsUrl(e.target.value)} placeholder="https://script.google.com/macros/s/.../exec" />
 </div>
 <div>
 <Label>Spreadsheet ID</Label>
 <Input value={sheetId} onChange={e => setSheetId(e.target.value)} placeholder="1Abc..." />
 </div>
 <div>
 <Label>Sheet tab name</Label>
 <Input value={sheetName} onChange={e => setSheetName(e.target.value)} placeholder="LiftLog" />
 </div>
 </div>

<div className="rounded-xl border border-neutral-200 bg-neutral-50 p-3 text-sm text-neutral-700">
 <div className="font-medium">{configured ? 'Ready for manual push' : 'Not ready yet'}</div>
 <ul className="mt-2 space-y-1 text-neutral-600 list-disc pl-5">
 <li>Lift Log saves locally first.</li>
 <li>Push copies your current local data into the sheet.</li>
 <li>There is no spreadsheet pull back into the app.</li>
 </ul>
 </div>

<div className="flex flex-col sm:flex-row gap-2">
 <Button variant="outline" onClick={saveGsConfig} className="sm:flex-1">Save sheet settings</Button>
 <Button onClick={pushToGoogleSheets} disabled={busy} className="sm:flex-1">
 {busy ? 'Pushing…' : 'Push local data to sheet'}
 </Button>
 </div>
 </Card>

<Card className="p-4 space-y-3">
 <div>
 <h2 className="text-base font-semibold">Backup and restore</h2>
 <p className="text-sm text-neutral-600 mt-1">Manual full backup export and manual import.</p>
 </div>
 <div className="flex flex-col sm:flex-row gap-2">
 <Button variant="outline" onClick={() => exportBackup(db)} className="sm:flex-1">{I.download} Export full backup</Button>
 <Button variant="outline" onClick={() => fileRef.current?.click()} className="sm:flex-1">{I.upload} Import backup file</Button>
 <input ref={fileRef} type="file" accept="application/json,.json" className="hidden" onChange={e => onImportFile(e.target.files?.)} />
 </div>
 </Card>

<Card className="p-4 space-y-3">
 <div>
 <h2 className="text-base font-semibold">Bulk import</h2>
 <p className="text-sm text-neutral-600 mt-1">Paste multiple day and exercise entries at once. This stays local and keeps the core import flow working.</p>
 </div>

<Textarea
 rows={10}
 value={bulkText}
 onChange={e => { setBulkText(e.target.value); setBulkPreview(null); setBulkError(''); }}
 placeholder={"DAY: Push Day\nGROUP: Upper\nEXERCISE: Bench Press | sets=3 | reps=8 | weight=60kg\nEXERCISE: Fly | sets=3 | reps=12"}
 className="font-mono text-xs"
 />

{bulkError && <div className="text-xs text-red-600 bg-red-50 border border-red-200 rounded-lg p-2">{bulkError}</div>}

{bulkPreview && (
 <div className="rounded-xl border border-green-200 bg-green-50 p-3 text-sm text-green-800">
 <div className="font-medium">Ready to import</div>
 <div className="mt-1 text-green-700">{bulkPreview.rows.length} exercise entries across {bulkPreview.dayNames.length} day(s).</div>
 </div>
 )}

<div className="flex flex-col sm:flex-row gap-2">
 {!bulkPreview && <Button variant="outline" onClick={parseNow} disabled={!bulkText.trim()} className="sm:flex-1">Preview import</Button>}
 {bulkPreview && <Button onClick={applyNow} className="sm:flex-1">Import now</Button>}
 </div>
 </Card>

<Card className="p-4 border-red-200 bg-red-50">
 <div className="space-y-3">
 <div>
 <h2 className="text-base font-semibold text-red-700">Reset app</h2>
 <p className="text-sm text-red-600 mt-1">This clears local data in this browser only.</p>
 </div>
 <Button variant="danger" onClick={resetAll}>Reset local app data</Button>
 </div>
 </Card>
 </div>
 {confirmNode}
 </PageShell>
 );
 }

function AppRouter() {
 const { path } = useHashRoute();

if (path === '/') return <HomePage />;
 if (path === '/groups') return <GroupsPage />;
 if (path === '/exercises') return <ExercisesPage />;
 if (path === '/history') return <HistoryPage />;
 if (path.startsWith('/history/')) return <HistoryDetailPage id={path.split('/')} />;
 if (path === '/calendar') return <CalendarPage />;
 if (path === '/settings') return <SettingsPage />;
 if (path.startsWith('/day/')) return <DayEditorPage id={path.split('/')} />;
 if (path.startsWith('/session/')) return <SessionPage id={path.split('/')} />;

return <PageShell title="Not found"><div className="p-4 text-sm text-neutral-500">Page not found.</div></PageShell>;
 }

function App() {
 return (
 <ErrorBoundary>
 <AppProvider>
 <div className="h-full flex flex-col bg-neutral-50">
 <div className="flex-1 min-h-0">
 <AppRouter />
 </div>
 <BottomNav />
 </div>
 </AppProvider>
 </ErrorBoundary>
 );
 }

const root = ReactDOM.createRoot(document.getElementById('root'));
